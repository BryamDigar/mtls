# --- PASO 1: Crear directorios para certificados ---
mkdir -p /etc/nginx/test/ca /etc/nginx/test/server /etc/nginx/test/client

# --- PASO 2: Generar CA Raíz (Autoridad de Certificación) ---
# Archivo de configuración para extensiones de CA
echo -e "[ v3_ca ]\nbasicConstraints = critical, CA:true\nkeyUsage = critical, keyCertSign, cRLSign\nsubjectKeyIdentifier = hash" > /etc/nginx/test/ca/ca-ext.cnf

# Generar clave privada de la CA
openssl genpkey -algorithm RSA -out /etc/nginx/test/ca/ca.key

# Generar certificado autofirmado de la CA (válido por 10 años)
openssl req -new -x509 -days 3650 -key /etc/nginx/test/ca/ca.key -out /etc/nginx/test/ca/ca.crt -subj "/C=CO/ST=Bogota/L=Bogota/O=BancoFachero/CN=BancoFachero-Root-CA/emailAddress=ca@bancofachero.com" -extensions v3_ca -config <(cat /etc/ssl/openssl.cnf <(echo -e "\n[ v3_ca ]\n$(cat /etc/nginx/test/ca/ca-ext.cnf)"))

# --- PASO 3: Generar Certificado de Servidor (para Nginx) ---
# Generar clave privada del servidor
openssl genpkey -algorithm RSA -out /etc/nginx/test/server/server.key

# Generar solicitud de firma de certificado (CSR) del servidor
openssl req -new -key /etc/nginx/test/server/server.key -out /etc/nginx/test/server/server.csr -subj "/C=CO/ST=Bogota/L=Bogota/O=BancoFachero/CN=servidor.bancofachero/emailAddress=server@bancofachero.com"

# Firmar el CSR del servidor con la CA raíz (válido por 1 año)
openssl x509 -req -in /etc/nginx/test/server/server.csr -CA /etc/nginx/test/ca/ca.crt -CAkey /etc/nginx/test/ca/ca.key -CAcreateserial -out /etc/nginx/test/server/server.crt -days 365

# --- PASO 4: Generar Certificado de Cliente (para Spring Boot) ---
# Generar clave privada del cliente
openssl genpkey -algorithm RSA -out /etc/nginx/test/client/client.key

# Generar solicitud de firma de certificado (CSR) del cliente
openssl req -new -key /etc/nginx/test/client/client.key -out /etc/nginx/test/client/client.csr -subj "/C=CO/ST=Bogota/L=Bogota/O=BancoFachero/CN=cliente1.bancofachero/emailAddress=cliente@bancofachero.com"

# Firmar el CSR del cliente con la CA raíz (válido por 1 año)
openssl x509 -req -in /etc/nginx/test/client/client.csr -CA /etc/nginx/test/ca/ca.crt -CAkey /etc/nginx/test/ca/ca.key -CAcreateserial -out /etc/nginx/test/client/client.crt -days 365

# --- PASO 5: Exportar Certificado de Cliente a formato PKCS#12 (.p12) para Java ---
# ¡IMPORTANTE! Te pedirá una contraseña para el archivo .p12. ¡Recuérdala!
# Este alias "cliente1.bancofachero" es el que usarás en Spring Boot.
openssl pkcs12 -export -inkey /etc/nginx/test/client/client.key -in /etc/nginx/test/client/client.crt -certfile /etc/nginx/test/ca/ca.crt -out /etc/nginx/test/client/client.p12 -name "cliente1.bancofachero"
